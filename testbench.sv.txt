
`timescale 1ns/1ps

module tb_alu;

// ------- Declare signals -------
reg  [3:0] A, B;
reg  [2:0] opcode;
wire [3:0] result;
wire       carry_out;
  
  alu_4bit uut (
    .A (A),
    .B (B),
    .opcode (opcode),
    .result (result),
    .carry_out (carry_out)
  );
  
initial begin
    $dumpfile("dump.vcd");
    $dumpvars(0, tb_alu);
end
  
integer pass_count, fail_count;
 
  task check;
    input[3:0] expected_result;
    input expected_carry;
    input[31:0] test_id;
    begin
        #10; 
        if (result === expected_result && carry_out === expected_carry) begin
            $display("PASS [%0d] | opcode=%b A=%0d B=%0d | result=%0d carry=%b",
                      test_id, opcode, A, B, result, carry_out);
            pass_count = pass_count + 1;
        end else begin
            $display("FAIL [%0d] | opcode=%b A=%0d B=%0d | got result=%0d carry=%b | expected result=%0d carry=%b",
                      test_id, opcode, A, B, result, carry_out, expected_result, expected_carry);
            fail_count = fail_count + 1;
        end
    end
endtask

  initial begin
    pass_count = 0;
    fail_count = 0;

    // ---- ADD (opcode 000) ----
    opcode=3'b000; A=4'd3;  B=4'd5;  check(4'd8,  1'b0, 1);  // 3+5=8
    opcode=3'b000; A=4'd9;  B=4'd7;  check(4'd0,  1'b1, 2);  // 9+7=16 → overflow
    opcode=3'b000; A=4'd0;  B=4'd0;  check(4'd0,  1'b0, 3);  // 0+0=0

    // ---- SUB (opcode 001) ----
    opcode=3'b001; A=4'd8;  B=4'd3;  check(4'd5,  1'b1, 4);  // 8-3=5
    opcode=3'b001; A=4'd5;  B=4'd5;  check(4'd0,  1'b1, 5);  // 5-5=0
    opcode=3'b001; A=4'd2;  B=4'd4;  check(4'd14, 1'b0, 6);  // 2-4 wraps (unsigned)

    // ---- AND (opcode 010) ----
    opcode=3'b010; A=4'b1100; B=4'b1010; check(4'b1000, 1'b0, 7);
    opcode=3'b010; A=4'b1111; B=4'b0000; check(4'b0000, 1'b0, 8);
    opcode=3'b010; A=4'b1111; B=4'b1111; check(4'b1111, 1'b0, 9);

    // ---- OR (opcode 011) ----
    opcode=3'b011; A=4'b1100; B=4'b0011; check(4'b1111, 1'b0, 10);
    opcode=3'b011; A=4'b0000; B=4'b0000; check(4'b0000, 1'b0, 11);
    opcode=3'b011; A=4'b1010; B=4'b0101; check(4'b1111, 1'b0, 12);

    // ---- NOT (opcode 100) ----
    opcode=3'b100; A=4'b0000; B=4'bx;   check(4'b1111, 1'b0, 13);
    opcode=3'b100; A=4'b1111; B=4'bx;   check(4'b0000, 1'b0, 14);
    opcode=3'b100; A=4'b1010; B=4'bx;   check(4'b0101, 1'b0, 15);

    // ---- Summary ----
    $display("----------------------------------------");
    $display("Results: %0d PASSED, %0d FAILED", pass_count, fail_count);
    $display("----------------------------------------");

    $finish;
end

endmodule
